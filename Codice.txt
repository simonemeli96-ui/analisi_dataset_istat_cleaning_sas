/* Adesso creiamo delle macro per poter visualizzare le frequenze e i grafici delle distribuzioni di frequenza
per gestire le variabili binarie*/
%macro analizza_binarie(dataset, lista_variabili);
	%let i = 1;

	%do %while(%scan(&lista_variabili., &i.) ne);
		%let var = %scan(&lista_variabili., &i.);
		title "Analisi della variabile &var";

		/* Frequenza e distribuzione */
		proc freq data=&dataset;
			tables &var / missing;
		run;

		title "Grafico a barre della variabile &var";

		proc sgplot data=&dataset;
			vbar &var / stat=freq;
		run;

		%let i = %eval(&i. + 1);
	%end;
%mend;

/* La macro seguente permette di creare delle tabelle di contingenza e dei grafici con frequenze relative*/
%macro tabella_grafico(dataset, var1, var2);
	proc freq data=&dataset noprint;
		tables &var1 * &var2 / out=tabella_contingenza;
	run;

	/* Calcolo del totale dei conteggi  */
	data _null_;
		retain totale 0;
		set tabella_contingenza end=ultimo;
		totale + count;

		if ultimo then
			call symput('totale', totale);
	run;

	data tabella_contingenza_pct;
		set tabella_contingenza;
		percent=(count / &totale) * 100;
	run;

	/* Creazione del grafico */
	proc sgplot data=tabella_contingenza_pct;
		vbar &var2 / response=percent group=&var1 groupdisplay=cluster datalabel;
		yaxis label="Frequenza relativa (%)";
		title "Distribuzione di &var2 per &var1";
	run;

%mend tabella_grafico;

	lsmeans CONDMi*ISTRMi / pdiff adjust=tukey;
